import pandas as pd
import numpy as np

# 1. Load data and dictionary
df = pd.read_csv('marketing_data.csv')
# Note: If your file uses a different delimiter like tab, use sep='\t'

print("Initial Shape:", df.shape)
print(df.info())

# 2. Handle Missing Values
# Example: Fill missing income with median
if 'Income' in df.columns:
    df['Income'] = df['Income'].fillna(df['Income'].median())

# Clean string columns if necessary (e.g., stripping spaces)
for col in df.select_dtypes(include=['object']).columns:
    df[col] = df[col].astype(str).str.strip()

# 3. Feature Engineering
# Calculate Age based on Year of Birth (Dt_Customer or Year_Birth)
current_year = 2026
if 'Year_Birth' in df.columns:
    df['Age'] = current_year - df['Year_Birth']

# Total Spend across product categories (Wine, Fruits, Meat, Fish, Sweets, Gold)
spend_cols = [c for c in ['MntWines', 'MntFruits', 'MntMeatProducts', 'MntFishProducts', 'MntSweetProducts', 'MntGoldProds'] if c in df.columns]
if spend_cols:
    df['Total_Spend'] = df[spend_cols].sum(axis=1)

# Total Purchases across channels (Web, Catalog, Store)
purchase_cols = [c for c in ['NumWebPurchases', 'NumCatalogPurchases', 'NumStorePurchases'] if c in df.columns]
if purchase_cols:
    df['Total_Purchases'] = df[purchase_cols].sum(axis=1)

# Family size / Total Children
child_cols = [c for c in ['Kidhome', 'Teenhome'] if c in df.columns]
if child_cols:
    df['Total_Children'] = df[child_cols].sum(axis=1)

# Save cleaned dataset for SQL and Streamlit
df.to_csv('cleaned_marketing_data.csv', index=False)
print("Data cleaning completed. Saved as 'cleaned_marketing_data.csv'.")
